## ABSOLUTE RULES - MUST FOLLOW

### 1. CODE STRUCTURE
❌ NEVER define functions outside the handler
✅ ALWAYS define ALL functions INSIDE the handler

```python
# ❌ WRONG - Will NOT work
def helper_function():
    return "data"

def handler(**params):
    return helper_function()

# ✅ CORRECT - Will always work
def handler(**params):
    def helper_function():
        return "data"
    return helper_function()
```

### 2. IMPORTS
❌ NEVER use "from X import Y"
✅ ALWAYS use full module path

```python
# ❌ WRONG
from datetime import datetime
from urllib.parse import quote

# ✅ CORRECT
import datetime
import urllib.parse
# Then use: datetime.datetime.now(), urllib.parse.quote()
```

### 3. EXTERNAL DEPENDENCIES
- beautifulsoup4 → import bs4
- pillow → import PIL
- opencv-python → import cv2

### 4. ASYNC/AWAIT RULES
- NEVER use asyncio.run() in a handler - it will cause "event loop already running" error
- NEVER make the handler async
- For async libraries like Playwright: use sync versions or subprocess
- Example for Playwright: use playwright.sync_api.sync_playwright instead of async version

### 5. PARAMETER VALIDATION PATTERN
Always validate parameters at the start of your handler:
```python

```

### 6. ERROR HANDLING PATTERN
Always handle errors gracefully:
```python
def handler(**params):
    import requests
    
    try:
        response = requests.get(url, timeout=10)
        response.raise_for_status()
        data = response.json()
        return {{"success": True, "data": data}}
    except requests.RequestException as e:
        return {{"error": f"API request failed: {{str(e)}}"}}
    except Exception as e:
        return {{"error": f"Unexpected error: {{str(e)}}"}}
```

### 7. SUBPROCESS PATTERN FOR HEAVY DEPENDENCIES
For tools like Playwright, use subprocess isolation:
```python
def handler(**params):

    import json
    
    script = f"""
import json
# Your heavy dependency code here

"""

            return {{"error": f"Script failed: {{result.stderr}}"}}

        return {{"error": "Operation timed out"}}
```

### Pattern 5: File/Data Transformation Service
```python
def handler(**params):
    import json
    import base64
    
    # Get input data
    input_data = params.get('data', '')
    input_format = params.get('input_format', 'json').lower()
    output_format = params.get('output_format', 'json').lower()
    
    if not input_data:
        return {{"error": "data parameter is required"}}
    
    try:
        # Parse input based on format
        if input_format == 'json':
            parsed_data = json.loads(input_data)
        elif input_format == 'base64':
            parsed_data = base64.b64decode(input_data).decode('utf-8')
        else:
            parsed_data = input_data
        
        # Transform to output format
        if output_format == 'json':
            if isinstance(parsed_data, str):
                output = {{"data": parsed_data}}
            else:
                output = parsed_data
        elif output_format == 'base64':
            if isinstance(parsed_data, dict):
                parsed_data = json.dumps(parsed_data)
            output = {{"encoded": base64.b64encode(parsed_data.encode()).decode()}}
        else:
            output = {{"data": str(parsed_data)}}
        
        return {{
            "success": True,
            "transformed": output,
            "input_format": input_format,
            "output_format": output_format
        }}
    except Exception as e:
        return {{"error": f"Transformation failed: {{str(e)}}"}}
```

### Pattern 7: Multi-step Processing Service
```python
def handler(**params):
    import re
    import json
    
    text = params.get('text', '')
    operations = params.get('operations', ['clean', 'analyze'])
    
    if not text:
        return {{"error": "text parameter is required"}}
    
    # Define processing functions
    def clean_text(text):
        # Remove extra whitespace
        text = re.sub(r'\\s+', ' ', text)
        # Remove special characters
        text = re.sub(r'[^\\w\\s]', '', text)
        return text.strip()
    
    def analyze_text(text):
        words = text.split()
        return {{
            "word_count": len(words),
            "char_count": len(text),
            "unique_words": len(set(words)),
            "avg_word_length": sum(len(w) for w in words) / len(words) if words else 0
        }}
    
    try:
        result = {{"original": text}}
        
        # Apply operations in sequence
        processed_text = text
        
        if 'clean' in operations:
            processed_text = clean_text(processed_text)
            result['cleaned'] = processed_text
        
        if 'analyze' in operations:
            analysis = analyze_text(processed_text)
            result['analysis'] = analysis
        
        result['success'] = True
        result['operations_applied'] = operations
        
        return result
        
    except Exception as e:
        return {{
            "error": f"Processing failed: {{str(e)}}",
            "success": False
        }}
```
